cmake_minimum_required(VERSION 3.25)
project(NEX8 VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NESCLONE_BUILD_TESTS "Build unit tests" ON)
option(NESCLONE_BUILD_SCRIPTING "Build NESLogic scripting prototype" ON)
option(NESCLONE_BUILD_TOOLS "Build ROM and asset diagnostic tools" ON)
option(NEX8_BUILD_ENGINE "Build NEX8 host-side game engine and Studio project services" ON)
option(NEX8_BUILD_CELL8 "Build experimental CELL8 hardware model" ON)
option(NEX8_BUILD_RAMDB "Build NEX8 RAM-resident relational dataset runtime" ON)

add_library(nes_core
    src/apu/apu.cpp src/cartridge/ines.cpp src/cartridge/mapper_factory.cpp src/cartridge/nrom.cpp
    src/controller/controller.cpp src/cpu/cpu6502.cpp src/debugger/debugger.cpp src/debugger/disassembler.cpp
    src/frontend/input.cpp src/frontend/library_service.cpp src/frontend/settings_store.cpp src/frontend/xmb_model.cpp
    src/nes/bus.cpp src/nes/nes.cpp src/ppu/ppu.cpp src/video/framebuffer.cpp)
add_library(nex8_runtime ALIAS nes_core)
target_include_directories(nes_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(nes_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(nes_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(nesclone src/main.cpp)
target_link_libraries(nesclone PRIVATE nes_core)

if(NEX8_BUILD_ENGINE)
    add_library(nex8_engine
        src/engine/game_engine.cpp src/studio/project.cpp
        src/input/input_system.cpp src/input/input_actions.cpp src/input/input_profile.cpp)
    target_include_directories(nex8_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(nex8_engine PRIVATE /W4 /permissive-)
    else()
        target_compile_options(nex8_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endif()

if(NEX8_BUILD_RAMDB)
    add_library(nex8_ramdb
        src/ramdb/ram_value.cpp src/ramdb/ram_schema.cpp src/ramdb/ram_table.cpp
        src/ramdb/ram_database.cpp src/ramdb/ram_sql.cpp src/ramdb/ram_binary.cpp)
    target_include_directories(nex8_ramdb PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(nex8_ramdb PRIVATE /W4 /permissive-)
    else()
        target_compile_options(nex8_ramdb PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endif()

if(NEX8_BUILD_CELL8)
    add_library(cell8_model
        src/cell8/cell8_cpu.cpp src/cell8/cell8_gpu.cpp src/cell8/cell8_spu.cpp
        src/cell8/dma_controller.cpp src/cell8/cartridge_image.cpp)
    target_include_directories(cell8_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(NEX8_BUILD_RAMDB)
        target_sources(cell8_model PRIVATE src/cell8/cell8_db_service.cpp)
        target_link_libraries(cell8_model PUBLIC nex8_ramdb)
    endif()
    if(NESCLONE_BUILD_TOOLS)
        add_executable(cell8rom tools/cell8rom/main.cpp)
        target_link_libraries(cell8rom PRIVATE cell8_model)
        if(NEX8_BUILD_RAMDB)
            add_executable(n8db tools/n8db/main.cpp)
            target_link_libraries(n8db PRIVATE nex8_ramdb)
        endif()
    endif()
    if(MSVC)
        target_compile_options(cell8_model PRIVATE /W4 /permissive-)
    else()
        target_compile_options(cell8_model PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endif()

if(NESCLONE_BUILD_SCRIPTING)
    add_library(neslogic_runtime
        src/neslogic/value.cpp src/neslogic/lexer.cpp src/neslogic/parser.cpp src/neslogic/runtime.cpp
        src/neslogic/interpreter.cpp src/neslogic/logic.cpp)
    target_include_directories(neslogic_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(neslogic_runtime PRIVATE /W4 /permissive-)
    else()
        target_compile_options(neslogic_runtime PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
    add_executable(neslogic src/neslogic/main.cpp)
    target_link_libraries(neslogic PRIVATE neslogic_runtime)
endif()

if(NESCLONE_BUILD_TOOLS)
    add_executable(nesrom tools/romtool/main.cpp)
    target_link_libraries(nesrom PRIVATE nes_core)
    add_executable(inesutil tools/inesutil/main.cpp)
    target_link_libraries(inesutil PRIVATE nes_core)
    add_executable(chrtool tools/chrtool/main.cpp)
    target_link_libraries(chrtool PRIVATE nes_core)
    add_executable(namtool tools/namtool/main.cpp)
    add_executable(nesdis tools/nesdis/main.cpp)
    target_link_libraries(nesdis PRIVATE nes_core)
endif()

    enable_testing()
    add_executable(nes_core_tests tests/test_main.cpp)
    target_link_libraries(nes_core_tests PRIVATE nes_core)

    add_executable(nes_frontend_tests tests/frontend_tests.cpp)
    target_link_libraries(nes_frontend_tests PRIVATE nes_core)
    add_test(NAME nes_frontend_tests COMMAND nes_frontend_tests)
    if(NESCLONE_BUILD_SCRIPTING)
        add_executable(neslogic_tests tests/neslogic_tests.cpp)
        target_link_libraries(neslogic_tests PRIVATE neslogic_runtime)
        add_test(NAME neslogic_tests COMMAND neslogic_tests)
    endif()
    if(NEX8_BUILD_ENGINE)
        add_executable(nex8_engine_tests tests/engine_tests.cpp)
        target_link_libraries(nex8_engine_tests PRIVATE nex8_engine)
        add_test(NAME nex8_engine_tests COMMAND nex8_engine_tests)
        add_executable(nex8_object_tests tests/engine_objects_tests.cpp)
        target_link_libraries(nex8_object_tests PRIVATE nex8_engine)
        add_test(NAME nex8_object_tests COMMAND nex8_object_tests)
        add_executable(nex8_input_tests tests/input_tests.cpp)
        target_link_libraries(nex8_input_tests PRIVATE nex8_engine nes_core)
        add_test(NAME nex8_input_tests COMMAND nex8_input_tests)
        add_executable(nex8_input_actions_tests tests/input_actions_tests.cpp)
        target_link_libraries(nex8_input_actions_tests PRIVATE nex8_engine nes_core)
        add_test(NAME nex8_input_actions_tests COMMAND nex8_input_actions_tests)
        add_executable(nex8_input_profile_tests tests/input_profile_tests.cpp)
        target_link_libraries(nex8_input_profile_tests PRIVATE nex8_engine nes_core)
        add_test(NAME nex8_input_profile_tests COMMAND nex8_input_profile_tests)
    endif()
    if(NEX8_BUILD_RAMDB)
        add_executable(nex8_ramdb_tests tests/ramdb_tests.cpp)
        target_link_libraries(nex8_ramdb_tests PRIVATE nex8_ramdb)
        add_test(NAME nex8_ramdb_tests COMMAND nex8_ramdb_tests)
        if(NEX8_BUILD_CELL8)
            add_executable(cell8_db_tests tests/cell8_db_service_tests.cpp)
            target_link_libraries(cell8_db_tests PRIVATE cell8_model)
            add_test(NAME cell8_db_tests COMMAND cell8_db_tests)
        endif()
    endif()
    if(NEX8_BUILD_CELL8)
        add_executable(cell8_tests tests/cell8_tests.cpp)
        target_link_libraries(cell8_tests PRIVATE cell8_model)
        add_test(NAME cell8_tests COMMAND cell8_tests)
    endif()
endif()
